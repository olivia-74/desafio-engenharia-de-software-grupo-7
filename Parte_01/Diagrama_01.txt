@startuml
autonumber
skinparam style strictuml

actor       "Usuário"               as U
boundary    "TelaRenovacao"         as TR
control     "RenovacaoController"   as RC
entity      "Emprestimo"            as E
entity      "Livro"                 as L
boundary    "APIPagamentoPix"       as PIX

U -> TR : solicitaRenovacaoEmprestimo()
activate TR
TR -> RC : processarRenovacao()
activate RC

RC -> E : obterLivro()
activate E
E --> RC : livro
deactivate E

RC -> L : verificaDisponibilidade()
activate L
L --> RC : possuiReserva (boolean)
deactivate L

alt Livro possui reserva
    RC --> TR : notificarImpedimento("Livro possui reserva")
    TR --> U : "Possui Reserva - Renovação Cancelada"

else Livro sem reserva
    RC -> E : verificaPrazo()
    activate E
    E --> RC : estaAtrasado (boolean)
    deactivate E
    
    alt Empréstimo em atraso
        RC --> TR : notificarImpedimentoPorAtraso()
        TR --> U : "Renovação bloqueada por atraso. Devolva o livro ou pague a multa para renovar."
        
        alt Usuário opta por pagar multa via Pix
            U -> TR : escolherPagamentoPix()
            TR -> RC : solicitarPix()
            RC -> E : calcularMultaAte(dataAtual)
            activate E
            E --> RC : valorMulta
            deactivate E
            RC -> PIX : gerarCodigoPix(valorMulta)
            activate PIX
            PIX --> RC : codigoPix
            deactivate PIX
            
            RC --> TR : exibirCodigoPix(codigoPix)
            TR --> U : exibeQRCodeEPix(codigoPix)
            
        else Usuário não escolhe pagar a multa
            TR --> U : "Realize a devolução do livro."
        end
        
    else Empréstimo em dia
        RC -> E : renovarEmprestimo()
        activate E
        E --> RC : novaDataDevolucao
        deactivate E
        RC --> TR : confirmarRenovacao(novaDataDevolucao)
        TR --> U : "Empréstimo renovado com sucesso!"
    end
end

deactivate RC
deactivate TR
@enduml
