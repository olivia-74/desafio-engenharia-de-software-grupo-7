@startuml Diagrama_03

class Livro {
  - titulo : String
  - autor : String
}

class Exemplar {
  - codigo : int
  - dimensoes : double[]
  - peso : double
}

class Usuario {
  - codigo : int
  - nome : String
}

class Endereco {
  - numero : int
  - rua : String
  - cidade : String
}

class Emprestimo {
  - dataRetirada : Date
  - dataEntregaPrevista : Date
  - dataEntregaRealizada : Date
  - estrategiaFrete : EstrategiaFrete
  + calcularFrete() : double
  + setEstrategiaFrete(estrategiaFrete : EstrategiaFrete) : void
}

interface EstrategiaFrete <<Strategy>> {
  + calcularFrete(endereco : Endereco, exemplar : Exemplar) : double
}

abstract class Empresa

class EmpresaA {
  + calcularFrete(endereco : Endereco, exemplar : Exemplar) : double
}

class EmpresaB {
  + calcularFrete(endereco : Endereco, exemplar : Exemplar) : double
}

class EmpresaC {
  + calcularFrete(endereco : Endereco, exemplar : Exemplar) : double
}

class EmpresaD {
  + calcularFrete(endereco : Endereco, exemplar : Exemplar) : double
}

Usuario "1" -- "0..*" Emprestimo : realiza

Usuario "1" -- "1" Endereco : possui

Emprestimo --> "0..1" EstrategiaFrete : usa

EstrategiaFrete <|.. Empresa
