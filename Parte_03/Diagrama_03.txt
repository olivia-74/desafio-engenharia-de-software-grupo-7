  - rua : String

class Emprestimo {
  - dataRetirada : Date
  - dataEntregaPrevista : Date
  - dataEntregaRealizada : Date
  - freteStrategy : CalculoFreteStrategy
  + calcularFrete() : double
  + setFreteStrategy(freteStrategy : CalculoFreteStrategy) : void
}

interface CalculoFreteStrategy <<Strategy>> {
  + calcular(endereco : Endereco, dimensoes : double[], peso : double) : double
}

  + calcular(endereco : Endereco, dimensoes : double[], peso : double) : double
